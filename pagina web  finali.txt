    function playSophisticatedStartupSound() {
      initAudioContext();
      if (!audioCtx) return;
      try {
        const now = audioCtx.currentTime;
        const freqs = [329.63, 493.88, 659.25, 987.77];
        freqs.forEach((freq, idx) => {
          const osc = audioCtx.createOscillator();
          const gainNode = audioCtx.createGain();
          const filter = audioCtx.createBiquadFilter();
          osc.type = idx % 2 === 0 ? 'sine' : 'triangle';
          osc.frequency.setValueAtTime(freq, now);
          filter.type = 'lowpass';
          filter.frequency.setValueAtTime(2500, now);
          gainNode.gain.setValueAtTime(0.0001, now);
          gainNode.gain.linearRampToValueAtTime(0.025, now + 0.15 + (idx * 0.05));
          gainNode.gain.exponentialRampToValueAtTime(0.00001, now + 2.2);
          osc.connect(filter);
          filter.connect(gainNode);
          gainNode.connect(audioCtx.destination);
          osc.start(now + (idx * 0.04));
          osc.stop(now + 2.4);
        });
      } catch (e) { console.warn("Startup sound error:", e); }
    }

    function playProfessionalSound(type = 'click', index = 0) {
      if (!isPlayingAudio) return;
      initAudioContext();
      if (!audioCtx) return;
      try {
        const now = audioCtx.currentTime;
        const osc = audioCtx.createOscillator();
        const gainNode = audioCtx.createGain();
        const filterNode = audioCtx.createBiquadFilter();
        const baseFreqs = [520, 580, 640, 680, 750, 820];
        let freq = baseFreqs[index % baseFreqs.length] || 500;
        if (type === 'hover') {
          osc.type = 'sine';
          osc.frequency.setValueAtTime(freq * 0.9, now);
          osc.frequency.exponentialRampToValueAtTime(freq, now + 0.04);
          gainNode.gain.setValueAtTime(0.001, now);
          gainNode.gain.linearRampToValueAtTime(0.025, now + 0.01);
          gainNode.gain.exponentialRampToValueAtTime(0.0001, now + 0.08);
        } else if (type === 'click') {
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(150, now);
          osc.frequency.exponentialRampToValueAtTime(80, now + 0.05);
          gainNode.gain.setValueAtTime(0.001, now);
          gainNode.gain.linearRampToValueAtTime(0.08, now + 0.01);
          gainNode.gain.exponentialRampToValueAtTime(0.0001, now + 0.1);
        } else if (type === 'release') {
          osc.type = 'sine';
          osc.frequency.setValueAtTime(800, now);
          osc.frequency.exponentialRampToValueAtTime(1200, now + 0.1);
          gainNode.gain.setValueAtTime(0.001, now);
          gainNode.gain.linearRampToValueAtTime(0.05, now + 0.02);
          gainNode.gain.exponentialRampToValueAtTime(0.0001, now + 0.3);
        } else if (type === 'copy') {
          osc.type = 'sine';
          osc.frequency.setValueAtTime(880, now);
          osc.frequency.exponentialRampToValueAtTime(1320, now + 0.08);
          gainNode.gain.setValueAtTime(0.001, now);
          gainNode.gain.linearRampToValueAtTime(0.06, now + 0.01);
          gainNode.gain.exponentialRampToValueAtTime(0.0001, now + 0.15);
        } else if (type === 'external') {
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(600, now);
          osc.frequency.exponentialRampToValueAtTime(950, now + 0.07);
          gainNode.gain.setValueAtTime(0.001, now);
          gainNode.gain.linearRampToValueAtTime(0.07, now + 0.01);
          gainNode.gain.exponentialRampToValueAtTime(0.0001, now + 0.12);
        }
        filterNode.type = 'lowpass';
        filterNode.frequency.value = 3200;
        osc.connect(filterNode);
        filterNode.connect(gainNode);
        gainNode.connect(audioCtx.destination);
        osc.start(now);
        osc.stop(now + 0.4);
      } catch (e) { console.warn("Audio Context error:", e); }
    }

    /* POOL DE AUDIO WOOSH - Reproduce completo sin interrumpir */
    const wooshPool = [];
    const WOOSH_POOL_SIZE = 12;
    let wooshIndex = 0;

    function initWooshPool() {
      const original = document.getElementById('woosh-sfx');
      if (!original) return;
      for (let i = 0; i < WOOSH_POOL_SIZE; i++) {
        const clone = original.cloneNode(true);
        clone.id = 'woosh-sfx-' + i;
        clone.removeAttribute('id');
        document.body.appendChild(clone);
        wooshPool.push(clone);
      }
    }

    function playWooshSound() {
      if (!isPlayingAudio) return;
      if (wooshPool.length === 0) return;
      const sfx = wooshPool[wooshIndex];
      wooshIndex = (wooshIndex + 1) % WOOSH_POOL_SIZE;
      if (sfx) {
        sfx.currentTime = 0;
        sfx.volume = 0.8;
        sfx.play().catch(e => console.warn("Woosh audio cancelado:", e));
      }
    }

    function copyToClipboard(text, btnElement, successText) {
      playProfessionalSound('copy');
      navigator.clipboard.writeText(text).then(() => {
        const originalHTML = btnElement.innerHTML;
        btnElement.classList.add("copied-glow");
        if (successText) {
          const span = btnElement.querySelector('span');
          if (span) span.innerText = successText;
          else btnElement.innerText = successText;
        }
        setTimeout(() => {
          btnElement.classList.remove("copied-glow");
          btnElement.innerHTML = originalHTML;
        }, 2500);
      }).catch(err => { console.error('No se pudo copiar el texto: ', err); });
    }

    /* ==========================================================================
       ANIMACIÓN DE TEXTOS POR LETRAS (SPLIT TEXT)
       ========================================================================== */
    function initSplitTitles() {
      document.querySelectorAll('[data-split-text]').forEach(titleEl => {
        if (!titleEl.querySelector('.title-char')) {
          const text = titleEl.textContent;
          titleEl.textContent = '';
          const words = text.split(' ');
          let globalIdx = 0;
          words.forEach((word, wIdx) => {
            const wordSpan = document.createElement('span');
            wordSpan.className = 'word-wrap';
            for (let i = 0; i < word.length; i++) {
              const charSpan = document.createElement('span');
              charSpan.className = 'title-char';
              charSpan.textContent = word[i];
              charSpan.style.transitionDelay = `${globalIdx * 0.025}s`;
              wordSpan.appendChild(charSpan);
              globalIdx++;
            }
            titleEl.appendChild(wordSpan);
            if (wIdx < words.length - 1) {
              const space = document.createTextNode(' ');
              titleEl.appendChild(space);
              globalIdx++;
            }
          });
        }
      });
    }

       ========================================================================== */
    function initMotionGraphicsEngine() {
      const interactiveElements = document.querySelectorAll('.mg-btn, .mg-btn-link');
      interactiveElements.forEach((el, index) => {
        el.addEventListener('mousemove', (e) => {
          const rect = el.getBoundingClientRect();
          const x = e.clientX - rect.left;
          const y = e.clientY - rect.top;
          const centerX = rect.width / 2;
          const centerY = rect.height / 2;
          const deltaX = (x - centerX) / centerX;
          const deltaY = (y - centerY) / centerY;
          const tiltX = deltaY * -10;
          const tiltY = deltaX * 10;
          const panX = deltaX * 4;
          const panY = deltaY * 4;
          el.style.transform = `perspective(800px) rotateX(${tiltX}deg) rotateY(${tiltY}deg) translate3d(${panX}px, ${panY}px, 0) scale(1.025)`;
          el.style.transition = 'transform 0.08s ease-out';
        });
        el.addEventListener('mouseenter', () => { playProfessionalSound('hover', index); });
        el.addEventListener('mouseleave', () => {
          el.style.transform = '';
          el.style.transition = 'transform 0.7s cubic-bezier(0.25, 1, 0.5, 1), background-color 0.6s cubic-bezier(0.25, 1, 0.5, 1), box-shadow 0.6s cubic-bezier(0.25, 1, 0.5, 1), border-color 0.6s cubic-bezier(0.25, 1, 0.5, 1)';
          el.classList.remove('is-pressed');
        });
        el.addEventListener('mousedown', () => {
          el.classList.add('is-pressed');
          el.classList.remove('is-released');
          playProfessionalSound('click');
        });
        el.addEventListener('mouseup', () => {
          if (el.classList.contains('is-pressed')) {
            el.classList.remove('is-pressed');
            el.classList.add('is-released');
            playProfessionalSound('release');
            setTimeout(() => { el.classList.remove('is-released'); }, 600);
            createRippleEffect(el);
            createMicroParticles(el);
          }
        });
      });
    }

    function createRippleEffect(element) {
      const ripple = document.createElement('div');
      ripple.classList.add('mg-ripple-ring');
      element.appendChild(ripple);
      setTimeout(() => { ripple.remove(); }, 600);
    }

    function createMicroParticles(element) {
      const particleCount = 4 + Math.floor(Math.random() * 3);
      const shapes = ['●', '＋', '▲', '◆'];
      for (let i = 0; i < particleCount; i++) {
        const particle = document.createElement('span');
        particle.classList.add('mg-particle-fx');
        particle.innerText = shapes[Math.floor(Math.random() * shapes.length)];
        const angle = Math.random() * Math.PI * 2;
        const velocity = 40 + Math.random() * 40;
        const tx = Math.cos(angle) * velocity;
        const ty = Math.sin(angle) * velocity - 20;
        const rot = (Math.random() - 0.5) * 360;
        particle.style.setProperty('--p-tx', `${tx}px`);
        particle.style.setProperty('--p-ty', `${ty}px`);
        particle.style.setProperty('--p-rot', `${rot}deg`);
        particle.style.left = `calc(50% - 6px)`;
        particle.style.top = `calc(50% - 6px)`;
        element.appendChild(particle);
        setTimeout(() => { particle.remove(); }, 500);
      }
    }

    function initLevelSounds() {
      const levels = document.querySelectorAll('.interactive-level');
      const levelFreqs = {
        'basic': 440,
        'intermediate': 554,
        'advanced': 659,
        'pro': 880
      };
      levels.forEach((el) => {
        const level = el.dataset.level;
        el.addEventListener('mouseenter', () => {
          if (!isPlayingAudio) return;
          initAudioContext();
          if (!audioCtx) return;
          try {
            const now = audioCtx.currentTime;
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            const filter = audioCtx.createBiquadFilter();
            const freq = levelFreqs[level] || 520;
            osc.type = 'sine';
            osc.frequency.setValueAtTime(freq, now);
            osc.frequency.exponentialRampToValueAtTime(freq * 1.5, now + 0.15);
            filter.type = 'lowpass';
            filter.frequency.value = 3000;
            gain.gain.setValueAtTime(0.0001, now);
            gain.gain.linearRampToValueAtTime(0.04, now + 0.02);
            gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.2);
            osc.connect(filter);
            filter.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start(now);
            osc.stop(now + 0.25);
          } catch (e) {}
        });
        el.addEventListener('mousedown', () => { playProfessionalSound('click'); });
      });
    }

    function openTab(evt, tabName) {
      playWooshSound();
      const tabContents = document.getElementsByClassName("tab-content");
      const tabBtns = document.getElementsByClassName("tab-btn");
      const currentBtn = evt.currentTarget;
      for (let i = 0; i < tabBtns.length; i++) { tabBtns[i].classList.remove("active"); }
      for (let i = 0; i < tabContents.length; i++) { tabContents[i].classList.remove("active"); }
      currentBtn.classList.add("active");
      setTimeout(() => {
        for (let i = 0; i < tabContents.length; i++) { tabContents[i].classList.remove("active-display"); }
        const selectedTab = document.getElementById(tabName);
        if (selectedTab) {
          selectedTab.classList.add("active-display");
          setTimeout(() => {
            selectedTab.classList.add("active");
            selectedTab.querySelectorAll('.motion-panel-intro').forEach(el => { el.classList.add('visible'); });
          }, 30);
        }
      }, 300);
    }

          musicToggleBtn.innerHTML = "🔇 Sonido Silenciado";
          musicToggleBtn.style.borderColor = "#ff4d4d";
          musicToggleBtn.style.color = "#ff8080";
          isPlayingAudio = false;
        } else {
          if (customRobloxAudio && customRobloxAudio.paused) customRobloxAudio.play();
          musicToggleBtn.innerHTML = "🎵 Sonido Activo";
          musicToggleBtn.style.borderColor = "#8b4513";
          musicToggleBtn.style.color = "#e6bc5c";

      });

        link.addEventListener('click', () => { playProfessionalSound('external'); });
      });

      const glow = document.getElementById('mouse-glow');
      const scrollContent = document.getElementById('scroll-content');
      let mouseX = window.innerWidth / 2;
      let mouseY = window.innerHeight / 2;
      let currX = mouseX;
      let currY = mouseY;

      document.addEventListener('mousemove', (e) => {
        mouseX = e.clientX;
        mouseY = e.clientY;
        const xPercent = (e.clientX / window.innerWidth) * 100;
        document.documentElement.style.setProperty('--mouse-x', `${xPercent}%`);
      });

      function renderFrame() {
        currX += (mouseX - currX) * 0.05;
        currY += (mouseY - currY) * 0.05;
        glow.style.left = `${currX}px`;
        glow.style.top = `${currY}px`;
        const moveX = (window.innerWidth / 2 - currX) * 0.025;
        const moveY = (window.innerHeight / 2 - currY) * 0.025;
        if (scrollContent) { scrollContent.style.transform = `translate3d(${moveX}px, ${moveY}px, 0px)`; }
        requestAnimationFrame(renderFrame);
      }
      requestAnimationFrame(renderFrame);

      const observerOptions = { root: null, rootMargin: '0px', threshold: 0.1 };
      const observer = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
          if (entry.isIntersecting) { entry.target.classList.add('visible'); }
          else { entry.target.classList.remove('visible'); }
        });
      }, observerOptions);
      document.querySelectorAll('.motion-panel-intro, [data-split-text]').forEach((el) => { observer.observe(el); });
    });
